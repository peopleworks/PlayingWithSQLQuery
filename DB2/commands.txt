-- DB2 administration cheatsheet.
-- Replace placeholder values before execution and keep credentials outside source control.

-- Attach user information to the current session.
ATTACH TO DB2 USER <DB2_USER> USING <DB2_PASSWORD>

-- Connect to a database.
CONNECT TO <DATABASE_NAME> USER <DB2_USER> USING <DB2_PASSWORD>

-- List all databases registered on the server.
LIST DATABASE DIRECTORY

-- List all tables inside a schema.
LIST TABLES FOR SCHEMA <SCHEMA_NAME>

-- Backup a database.
BACKUP DATABASE <DATABASE_NAME> TO <BACKUP_PATH>

-- Restore a database from a specific backup image.
RESTORE DATABASE <DATABASE_NAME> FROM <BACKUP_PATH> TAKEN AT <BACKUP_TIMESTAMP>

-- List all connected applications.
LIST APPLICATIONS

-- Close all connected applications.
FORCE APPLICATIONS ALL

-- List active databases.
LIST ACTIVE DATABASES

-- List all applications connected to a database with details.
LIST APPLICATIONS FOR DATABASE <DATABASE_NAME> SHOW DETAIL

-- Get the current working instance.
GET INSTANCE
